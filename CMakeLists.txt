cmake_minimum_required(VERSION 3.20)
project(socom_native_tools LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

function(add_socom_tool name source)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${source}")
        add_executable(${name} ${source})
        if(MSVC)
            target_compile_options(${name} PRIVATE /W4)
        else()
            target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endif()
endfunction()

add_socom_tool(zar_catalog tools/archive/zar_catalog.cpp)
add_socom_tool(models_probe tools/models/models_probe.cpp)
add_socom_tool(socom_rigid_gltf tools/models/socom_rigid_gltf.cpp)
add_socom_tool(socom_character_gltf tools/models/socom_character_gltf.cpp)
add_socom_tool(socom_zed_texture_extract tools/textures/socom_zed_texture_extract.cpp)
add_socom_tool(socom_texture_tool tools/textures/socom_texture_tool.cpp)
add_socom_tool(socom_bone_registry tools/animation/socom_bone_registry.cpp)
