cmake_minimum_required(VERSION 3.20)
project(socom_cseal_recovery LANGUAGES CXX)
add_executable(recover_cseal recover_cseal.cpp)
target_compile_features(recover_cseal PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(recover_cseal PRIVATE /W4 /permissive-)
else()
  target_compile_options(recover_cseal PRIVATE -Wall -Wextra -Wpedantic)
endif()
