cmake_minimum_required(VERSION 3.20)
project(socom_recomp_recovery_smoke LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(socom_recomp_recovery_smoke recomp_recovery_smoke.cpp)
target_include_directories(socom_recomp_recovery_smoke PRIVATE "${CMAKE_CURRENT_LIST_DIR}" "${CMAKE_CURRENT_LIST_DIR}/..")
if(MSVC)
  target_compile_options(socom_recomp_recovery_smoke PRIVATE /W4 /permissive-)
else()
  target_compile_options(socom_recomp_recovery_smoke PRIVATE -Wall -Wextra -Wpedantic)
endif()
