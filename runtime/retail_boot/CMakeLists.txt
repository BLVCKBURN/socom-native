cmake_minimum_required(VERSION 3.20)
project(socom_retail_boot LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "socom_retail_boot is a Windows target.")
endif()
add_executable(socom_retail_boot WIN32 main.cpp)
target_compile_features(socom_retail_boot PRIVATE cxx_std_17)
target_compile_definitions(socom_retail_boot PRIVATE UNICODE _UNICODE)
target_link_libraries(socom_retail_boot PRIVATE shell32)
if(MSVC)
    target_compile_options(socom_retail_boot PRIVATE /W4 /permissive-)
endif()
