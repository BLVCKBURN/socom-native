cmake_minimum_required(VERSION 3.20)
project(socom_retail_menu_v3 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows target")
endif()
add_executable(socom_retail_menu_v3 WIN32 main.cpp)
target_compile_features(socom_retail_menu_v3 PRIVATE cxx_std_17)
target_compile_definitions(socom_retail_menu_v3 PRIVATE UNICODE _UNICODE)
target_link_libraries(socom_retail_menu_v3 PRIVATE shell32)
if(MSVC)
  target_compile_options(socom_retail_menu_v3 PRIVATE /W4 /permissive-)
endif()
